% Beamer template for the Helsinki Logic Seminar talk, 11 November 2026.
%
% Build (the PDF lands here; auxiliary files go to the gitignored build/ folder):
%   latexmk -pdf -auxdir=build talk_template.tex
%
% Present (presenter notes on the laptop, slides in their own window):
%   pdfpc -w both -n right talk_template.pdf
%
% The companion talk_template.pdf is this file compiled as it stands, so its pages
% are double-width: slide on the left, notes on the right. That is the version to
% present from. To circulate the slides, comment out the notes option below.

\documentclass[aspectratio=169,11pt]{beamer}

\usetheme{default}
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{footline}[frame number]

% Doubles every page: slide on the left, notes on the right. pdfpc needs this
% for -n right. Comment it out to build the version you circulate.
\setbeameroption{show notes on second screen=right}

\title{Your title}
\author{Your name}
\date{Helsinki Logic Seminar, 11 November 2026}

\begin{document}

\begin{frame}
  \titlepage
\end{frame}

\begin{frame}{Frame title}
  \begin{itemize}
    \item<1-> Shown from the first click
    \item<2-> Appears on the second click
  \end{itemize}

  \begin{definition}
    A map $f : A \to B$ is an equivalence if \ldots
  \end{definition}

  \note{What I say here, visible only to me.}
\end{frame}

\end{document}
